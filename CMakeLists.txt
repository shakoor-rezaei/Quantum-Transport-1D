cmake_minimum_required(VERSION 3.16)
project(quantum_transport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(qtransport src/distributions.cpp src/transport.cpp)
target_include_directories(qtransport PUBLIC include)
add_executable(quantum_transport src/main.cpp)
target_link_libraries(quantum_transport PRIVATE qtransport Threads::Threads)
enable_testing()
add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core PRIVATE qtransport Threads::Threads)
add_test(NAME numerical_checks COMMAND test_core)

find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME cli_reproducibility COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py" $<TARGET_FILE:quantum_transport>)
endif()
